# llama.cpp's src/ on the unified ggml, for the PC checks of the text engine (tools/test-llama/run.sh).
# GGML_DIR and LLAMA_DIR come from run.sh: scripts/ggml-unified.sh and scripts/llama-fetch.sh outputs.
cmake_minimum_required(VERSION 3.16)
project(paul3d_test_llama C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(${GGML_DIR} ggml)
file(GLOB LLAMA_SRC ${LLAMA_DIR}/src/*.cpp ${LLAMA_DIR}/src/models/*.cpp)
file(WRITE ${CMAKE_BINARY_DIR}/gen/llama-version.h "#pragma once\n#define LLAMA_VERSION \"paul3d\"\n#define LLAMA_COMMIT \"4ed1c85\"\n")
add_library(llama STATIC ${LLAMA_SRC})
target_include_directories(llama PUBLIC ${LLAMA_DIR}/include PRIVATE ${LLAMA_DIR}/src ${CMAKE_BINARY_DIR}/gen)
target_link_libraries(llama PUBLIC ggml)
add_executable(hello hello.cpp)
target_link_libraries(hello llama)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/test_text.cpp AND EXISTS ${PAUL3D_PS5}/mg_text.cpp)
  add_executable(test_text test_text.cpp ${PAUL3D_PS5}/mg_text.cpp)
  target_include_directories(test_text PRIVATE ${PAUL3D_PS5})
  target_link_libraries(test_text llama)
endif()
